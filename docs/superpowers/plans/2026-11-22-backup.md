> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. **Task 1 → opencode** (`opencode run --auto -m opencode/hy4-flash-free`, intern rules: verbatim code, stop on first failure). **Task 2 → Claude subagent on Sonnet.** Work happens in the git worktree `D:\claude-exp\wt-backup` (branch `feat/backup`); workers never commit; the main session reviews, merges and commits. Never give a worker `local.properties` or anything under `docs/Books-Photo/`.

**Goal:** Spec §6 — Settings → "Backup & restore": **Export** a `.zip` (all books, categories, links, notes, profile text + cover images) through the system file picker, and **Import** a backup by merging (newer `updatedAt` wins, so importing twice is harmless and it works as "move to a new phone"). Shows "Last exported …".

**Architecture:** Pure, unit-tested core (`backup/`): DTOs + mapping, `mergeNewer` rule, `BackupZip` (write/read with zip-slip protection). Android layer on top: DAO "all rows incl. deleted" queries, `BackupManager` (ContentResolver streams + Room transaction), a screen using `CreateDocument`/`OpenDocument`.

**Format v1** (`backup.json` + `covers/<bookId>.<ext>` in one zip): `{version:1, exportedAt, books[], categories[], bookCategories[], profile?{name,bio,yearlyGoal}}`. Soft-deleted rows are included (so deletions propagate). Not included: import queue, settings, avatar photo, AI models.

import org.junit.Assert.assertEquals
import org.junit.Test

class BackupMergeTest {
    private data class Row(val id: String, val updatedAt: Long)

    @Test fun newRowsAreWritten() =
        assertEquals(listOf(Row("a", 1)), mergeNewer(listOf(Row("a", 1)), emptyMap(), { it.id }, { it.updatedAt }))

    @Test fun newerWinsOlderAndEqualAreSkipped() {
        val existing = mapOf("a" to 5L, "b" to 5L, "c" to 5L)
        val incoming = listOf(Row("a", 6), Row("b", 5), Row("c", 4), Row("d", 1))
        assertEquals(listOf(Row("a", 6), Row("d", 1)), mergeNewer(incoming, existing, { it.id }, { it.updatedAt }))
    }

    @Test fun importingTheSameRowsTwiceWritesNothingTheSecondTime() {
        val rows = listOf(Row("a", 3), Row("b", 4))
        val afterFirst = mergeNewer(rows, emptyMap(), { it.id }, { it.updatedAt }).associate { it.id to it.updatedAt }
        assertEquals(emptyList<Row>(), mergeNewer(rows, afterFirst, { it.id }, { it.updatedAt }))
    }

    @Test fun summaryCounts() {
        val s = MergeSummary(added = 2, updated = 1, unchanged = 3)
        assertEquals(6, s.total)
    }
}

import java.io.IOException

import java.util.zip.ZipOutputStream

class BackupZipTest {
    @get:Rule val tmp = TemporaryFolder()

    private val backup = BackupFile(
        exportedAt = 7L,
        books = listOf(BookDto(id = "b1", title = "T", status = "WANT", updatedAt = 1L, coverFile = "covers/b1.jpg")),

        ByteArrayOutputStream().also { ZipOutputStream(it).use(write) }.toByteArray()

    @Test fun writeThenReadRoundTripsJsonAndCovers() {
        val cover = File(tmp.root, "src.jpg").apply { writeBytes(byteArrayOf(1, 2, 3, 4)) }
        val out = ByteArrayOutputStream()
        BackupZip.write(out, backup, mapOf("covers/b1.jpg" to cover))

    @Test fun missingCoverFilesAreSkippedOnWrite() {
        val out = ByteArrayOutputStream()
        BackupZip.write(out, backup, mapOf("covers/b1.jpg" to File(tmp.root, "does-not-exist.jpg")))
        val dest = File(tmp.root, "covers").apply { mkdirs() }
        BackupZip.read(ByteArrayInputStream(out.toByteArray()), dest)
        assertFalse(File(dest, "b1.jpg").exists())
    }

    @Test fun zipSlipEntriesAreIgnored() {
        val bytes = zipBytes { z ->
            z.putNextEntry(ZipEntry(BackupZip.JSON)); z.write(backupJson.encodeToString(BackupFile.serializer(), backup).toByteArray()); z.closeEntry()
            z.putNextEntry(ZipEntry("covers/../evil.txt")); z.write(9); z.closeEntry()
            z.putNextEntry(ZipEntry("covers/sub/dir.jpg")); z.write(9); z.closeEntry()
            z.putNextEntry(ZipEntry("other/thing.bin")); z.write(9); z.closeEntry()
        }
        val dest = File(tmp.root, "covers").apply { mkdirs() }
        BackupZip.read(ByteArrayInputStream(bytes), dest)
        assertTrue(dest.listFiles().orEmpty().isEmpty())
        assertFalse(File(tmp.root, "evil.txt").exists())

        assertThrows(IOException::class.java) { BackupZip.read(ByteArrayInputStream(bytes), tmp.root) }
    }

package com.exc33ded.library.backup

/** Rows to write: new ids, or ids whose incoming updatedAt is strictly newer than the stored one. */
fun <T> mergeNewer(incoming: List<T>, existingUpdatedAt: Map<String, Long>, idOf: (T) -> String, updatedOf: (T) -> Long): List<T> =
    incoming.filter { row -> existingUpdatedAt[idOf(row)].let { it == null || updatedOf(row) > it } }

data class MergeSummary(val added: Int, val updated: Int, val unchanged: Int) {
    val total: Int get() = added + updated + unchanged
}

`backup/BackupZip.kt`:
```kotlin
package com.exc33ded.library.backup

import java.io.File
import java.io.IOException
import java.io.InputStream
import java.io.OutputStream
import java.util.zip.ZipEntry
import java.util.zip.ZipInputStream
import java.util.zip.ZipOutputStream

object BackupZip {
    const val JSON = "backup.json"
    private const val COVERS = "covers/"

    /** [covers] maps a zip entry name ("covers/<id>.<ext>") to the file to store; missing files are skipped. */
    fun write(out: OutputStream, backup: BackupFile, covers: Map<String, File>) {
        ZipOutputStream(out).use { zip ->
            zip.putNextEntry(ZipEntry(JSON))
            zip.write(backupJson.encodeToString(BackupFile.serializer(), backup).toByteArray(Charsets.UTF_8))
            zip.closeEntry()
            covers.forEach { (name, file) ->
                if (file.isFile) {
                    zip.putNextEntry(ZipEntry(name))
                    file.inputStream().use { it.copyTo(zip) }
                    zip.closeEntry()
                }
            }
        }
    }

    /**
     * Parses backup.json and extracts cover images into [coversDir] (overwriting same-named files).
     * Only entries named exactly "covers/<plain file name>" are extracted; anything else (paths, "..") is ignored.
     */
    fun read(input: InputStream, coversDir: File): BackupFile {
        var json: String? = null
        ZipInputStream(input).use { zip ->
            while (true) {
                val entry = zip.nextEntry ?: break
                val name = entry.name
                when {
                    entry.isDirectory -> Unit
                    name == JSON -> json = zip.readBytes().toString(Charsets.UTF_8)
                    name.startsWith(COVERS) -> {
                        val leaf = name.removePrefix(COVERS)
                        if (leaf.isNotEmpty() && '/' !in leaf && '\\' !in leaf && leaf != ".." && leaf != ".") {
                            coversDir.mkdirs()
                            File(coversDir, leaf).outputStream().use { zip.copyTo(it) }
                        }
                    }
                }
            }
        }
        val text = json ?: throw IOException("This is not a library backup (backup.json is missing)")
        val backup = backupJson.decodeFromString(BackupFile.serializer(), text)
        if (backup.version > BACKUP_VERSION) throw IOException("This backup was made by a newer version of the app")
        return backup
    }
}
```
(Note `zip.readBytes()` on a `ZipInputStream` reads only the current entry — correct.)
